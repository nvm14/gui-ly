  <script>
    // --- DOM Elements ---
    const btnYes = document.getElementById('btnYes');
    const btnNo = document.getElementById('btnNo');
    const buttonArea = document.getElementById('buttonArea');
    const initialBtnGroup = document.getElementById('initialBtnGroup');
    const questionText = document.getElementById('questionText');
    const subHint = document.getElementById('subHint');
    const successCard = document.getElementById('successCard');
    const bunnySvgWrapper = document.getElementById('bunnySvgWrapper');
    const bunnyEyes = document.getElementById('bunnyEyes');
    const bunnyMouth = document.getElementById('bunnyMouth');
    const headHearts = document.getElementById('headHearts');
    const btnReplay = document.getElementById('btnReplay');

    // 10 dynamic funny texts for the runaway button
    const teasingTexts = [
      "Hông bé ơi 😜",
      "Kó cái giề mà Kó :3",
      "Ai cho mà bấm hả?",
      "Cấm từ chối nhaaa 🥺",
      "Nút này bị hỏng rùi!",
      "Mơ đii, hổng cho bấm đâu 😝",
      "Chạy đâu cho thoát nà",
      "Bấm CÓ lẹ lên điii ❤️",
      "Chỉ được chọn CÓ thui!",
      "Năn nỉ bấm Có mừ 🥰"
    ];

    let dodgeCount = 0;
    let yesScale = 1;
    let isSeparated = false;

    // --- Floating Background Elements ---
    function initFloatingBackground() {
      const container = document.getElementById('ambientContainer');
      const icons = ['🌸', '✨', '💖', '🍿', '🌹', '❤️'];
      const total = 18;

      for (let i = 0; i < total; i++) {
        const span = document.createElement('div');
        span.className = 'bg-particle text-xl sm:text-2xl select-none';
        span.textContent = icons[Math.floor(Math.random() * icons.length)];
        span.style.left = `${Math.random() * 95}vw`;
        span.style.animationDuration = `${6 + Math.random() * 6}s`;
        span.style.animationDelay = `${Math.random() * 5}s`;
        container.appendChild(span);
      }
    }
    initFloatingBackground();

    // --- Sound Effects with Web Audio API (Clean tone generator) ---
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    let audioContext = null;

    function playTone(freq, type = 'sine', duration = 0.12) {
      try {
        if (!audioContext) audioContext = new AudioContextClass();
        if (audioContext.state === 'suspended') audioContext.resume();

        const osc = audioContext.createOscillator();
        const gain = audioContext.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, audioContext.currentTime);
        gain.gain.setValueAtTime(0.12, audioContext.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioContext.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioContext.destination);
        osc.start();
        osc.stop(audioContext.currentTime + duration);
      } catch (e) {}
    }

    function playDodgeSfx() {
      playTone(400 + Math.random() * 150, 'sine', 0.1);
    }

    function playFanfare() {
      const notes = [523.25, 659.25, 783.99, 1046.50, 1318.51];
      notes.forEach((f, i) => {
        setTimeout(() => playTone(f, 'triangle', 0.28), i * 110);
      });
    }

    // --- The Core Mechanic: Dodge "Không" & Center "Có" ---
    function handleDodge(e) {
      if (e) {
        e.preventDefault();
        e.stopPropagation();
      }

      dodgeCount++;
      playDodgeSfx();

      // STEP 1: If first time touched, detach and place "Có" right at Center Stage
      if (!isSeparated) {
        isSeparated = true;

        // Make button area pure center anchor
        initialBtnGroup.classList.remove('justify-center', 'gap-6');
        initialBtnGroup.classList.add('relative', 'w-full', 'h-full');

        // Center the "Có" button right inside the buttonArea
        btnYes.style.position = 'absolute';
        btnYes.style.left = '50%';
        btnYes.style.top = '50%';
        btnYes.style.transform = 'translate(-50%, -50%) scale(1.1)';
        btnYes.style.zIndex = '30';
        btnYes.classList.add('heart-pulse');

        // Turn "Không" button into free-floating fixed viewport element
        btnNo.style.position = 'fixed';
        btnNo.style.zIndex = '40';
      }

      // STEP 2: Enlarge the centered "Có" button progressively
      yesScale += 0.12;
      btnYes.style.transform = `translate(-50%, -50%) scale(${yesScale})`;

      // STEP 3: Change text of "Không" to tease
      const textIdx = dodgeCount % teasingTexts.length;
      btnNo.textContent = teasingTexts[textIdx];

      // STEP 4: Relocate "Không" to a random location on screen away from center
      const btnW = btnNo.offsetWidth || 110;
      const btnH = btnNo.offsetHeight || 50;
      const padding = 16;

      const screenW = window.innerWidth;
      const screenH = window.innerHeight;

      let newX = Math.floor(Math.random() * (screenW - btnW - padding * 2)) + padding;
      let newY = Math.floor(Math.random() * (screenH - btnH - padding * 2)) + padding;

      // Ensure it doesn't overlap the centered "Có" button
      const yesBox = btnYes.getBoundingClientRect();
      if (
        newX < yesBox.right + 30 &&
        newX + btnW > yesBox.left - 30 &&
        newY < yesBox.bottom + 30 &&
        newY + btnH > yesBox.top - 30
      ) {
        newY = (newY > screenH / 2) ? padding + 20 : screenH - btnH - padding - 20;
      }

      btnNo.style.left = `${newX}px`;
      btnNo.style.top = `${newY}px`;

      // Update playful hint
      subHint.textContent = "Hihi nút CÓ ở giữa to đùng kìa, bấm đi màaa~ 💖";
      subHint.classList.remove('text-pink-600/80');
      subHint.classList.add('text-rose-600', 'animate-pulse');
    }

    // Attach interaction events: mouse hover, touch tap, mouse click
    btnNo.addEventListener('mouseenter', handleDodge);
    btnNo.addEventListener('pointerenter', handleDodge);
    btnNo.addEventListener('touchstart', handleDodge, { passive: false });
    btnNo.addEventListener('click', handleDodge);

    // --- Success Event when "Có" is Clicked ---
    function triggerCelebration() {
      // 1. Center heart firework
      confetti({
        particleCount: 100,
        spread: 100,
        origin: { y: 0.6 },
        colors: ['#ff4d6d', '#ff758f', '#ff8fa3', '#ffb3c1', '#ffe5ec']
      });

      // 2. Dual cannon side celebration
      const end = Date.now() + 2500;
      const interval = setInterval(() => {
        if (Date.now() > end) {
          clearInterval(interval);
          return;
        }
        confetti({
          particleCount: 3,
          angle: 60,
          spread: 60,
          origin: { x: 0, y: 0.65 },
          colors: ['#ff4d6d', '#ffccd5', '#ffffff']
        });
        confetti({
          particleCount: 3,
          angle: 120,
          spread: 60,
          origin: { x: 1, y: 0.65 },
          colors: ['#ff4d6d', '#ffccd5', '#ffffff']
        });
      }, 50);
    }

      playFanfare();
      triggerCelebration();

      bunnySvgWrapper.classList.remove('bunny-idle');
      bunnySvgWrapper.classList.add('bunny-happy');

        <path d="M 124 175 Q 134 163 144 175" stroke="#382823" stroke-width="6" stroke-linecap="round" fill="none"/>
        <path d="M 194 175 Q 204 163 214 175" stroke="#382823" stroke-width="6" stroke-linecap="round" fill="none"/>

        <path d="M 152 178 Q 169 170 186 178 C 188 212, 150 212, 152 178 Z" 
              fill="#ff4d6d" stroke="#382823" stroke-width="6" stroke-linejoin="round"/>
      `;

      // Floating celebratory emojis over bunny head
      headHearts.innerHTML = `
        <span class="text-rose-500 text-3xl animate-bounce">💖</span>
        <span class="text-pink-500 text-4xl animate-pulse">🥰</span>
        <span class="text-rose-500 text-3xl animate-bounce">🍿</span>
      `;

      // Hide question text and button playground
      questionText.classList.add('hidden');
      subHint.classList.add('hidden');
      buttonArea.classList.add('hidden');
      if (btnNo) btnNo.style.display = 'none';

      // Show and animate success confirmation card
      successCard.classList.remove('hidden');
      setTimeout(() => {
        successCard.classList.remove('opacity-0', 'translate-y-6');
      }, 50);
    }

    btnYes.addEventListener('click', onAccepted);

    // --- Reset to initial state ---
    btnReplay.addEventListener('click', () => {
      successCard.classList.add('opacity-0', 'translate-y-6');

      setTimeout(() => {
        successCard.classList.add('hidden');
        questionText.classList.remove('hidden');
        subHint.classList.remove('hidden');
        buttonArea.classList.remove('hidden');

        // Reset Bunny
        bunnySvgWrapper.classList.remove('bunny-happy');
        bunnySvgWrapper.classList.add('bunny-idle');
        bunnyEyes.innerHTML = `
          <circle cx="132" cy="172" r="7" fill="#382823" />
          <circle cx="204" cy="172" r="7" fill="#382823" />
        `;
        bunnyMouth.innerHTML = `
          <path d="M 154 180 Q 169 174 184 180 C 184 206, 154 206, 154 180 Z" 
                fill="#ffffff" stroke="#382823" stroke-width="6" stroke-linejoin="round" stroke-linecap="round"/>
        `;
        headHearts.innerHTML = `
          <span class="text-rose-400 text-2xl">✨</span>
          <span class="text-pink-500 text-3xl">🌸</span>
          <span class="text-rose-400 text-2xl">✨</span>
        `;

        // Reset Buttons
        dodgeCount = 0;
        yesScale = 1;
        isSeparated = false;

        initialBtnGroup.className = 'flex items-center justify-center gap-6 w-full';
        btnYes.style.position = '';
        btnYes.style.left = '';
        btnYes.style.top = '';
        btnYes.style.transform = '';
        btnYes.style.zIndex = '';
        btnYes.classList.remove('heart-pulse');

        btnNo.style.display = '';
        btnNo.style.position = '';
        btnNo.style.left = '';
        btnNo.style.top = '';
        btnNo.style.zIndex = '';
        btnNo.textContent = 'Kó';

        subHint.textContent = "(Em chỉ được chọn 1 trong 2 thui nhen! 🎬🍿)";
        subHint.classList.remove('text-rose-600', 'animate-pulse');
        subHint.classList.add('text-pink-600/80');
      }, 300);
    });
